1) Um funcionário de uma empresa recebe, anualmente, aumento salarial. Sabe-se que:
a) Esse funcionário foi contratado em 2005, com salário inicial de R$ 1.000,00.
b) Em 2006, ele recebeu aumento de 1,5% sobre seu salário inicial.
c) A partir de 2007 (inclusive), os aumentos salariais sempre corresponderam ao dobro do percentual do ano anterior.
Faça um programa que determine o salário atual desse funcionário.

ALGORITMO

DECLARE salario_inicial, salario_atual, percentual, ano_atual, i NUMERICO;

ESCREVA "INFORME O ANO ATUAL PARA O CALCULO:";

LEIA ano_atual;

SE ano_atual < 2006

ENTÃO INÍCIO

    ESCREVA "O ANO DEVE SER IGUAL OU SUPERIOR A 2006."

FIM

SENÃO INÍCIO

    salario_inicial <- 1000.00;
    percentual <- 0.015;

    salario_atual <- salario_inicial + (salario_inicial * percentual);

    PARA i DE 2007 ATÉ ano_atual FACA

        percentual <- percentual * 2;
        salario_atual <- salario_atual + (salario_atual * percentual);

    FIM_PARA

    ESCREVA "O SALARIO ATUAL DO FUNCIONARIO E: R$ ", salario_atual;

FIM

FIM_ALGORITMO

2) Foi feita uma estatística em cinco cidades brasileiras para coletar dados sobre acidentes de trânsito. Foram obtidos os seguintes dados:

a) código da cidade;

b) número de veículos de passeio;

c) número de acidentes de trânsito com vítimas.

Deseja-se saber:

a) qual é o maior e qual é o menor índice de acidentes de trânsito e a que cidades pertencem;

b) qual é a média de veículos nas cinco cidades juntas;

c) qual é a média de acidentes de trânsito nas cidades com menos de 2.000 veículos de passeio.

ALGORITMO

DECLARE CODIGO_CIDADE, ID_CIDADE_MAIOR, ID_CIDADE_MENOR,
        MAIOR_NUM_ACIDENTES, MENOR_NUM_ACIDENTES,
        NUM_VEICULOS, NUM_ACIDENTES,
        MEDIA_VEICULOS, MEDIA_ACIDENTES,
        TOTAL_VEICULOS, TOTAL_ACIDENTES, CIDADES_MENOS_2000 NUMERICO;

MAIOR_NUM_ACIDENTES <- -9999;
MENOR_NUM_ACIDENTES <- 99999;

TOTAL_VEICULOS <- 0;
TOTAL_ACIDENTES <- 0;
CIDADES_MENOS_2000 <- 0;

PARA CODIGO_CIDADE DE 1 ATÉ 5 FAÇA

INÍCIO

    ESCREVA "INSIRA O NÚMERO DE VEÍCULOS";
    LEIA NUM_VEICULOS;

    ESCREVA "INSIRA O NÚMERO DE ACIDENTES";
    LEIA NUM_ACIDENTES;

    SE MENOR_NUM_ACIDENTES > NUM_ACIDENTES ENTÃO

    INÍCIO
        MENOR_NUM_ACIDENTES <- NUM_ACIDENTES;
        ID_CIDADE_MENOR <- CODIGO_CIDADE;
    FIM

    SE MAIOR_NUM_ACIDENTES < NUM_ACIDENTES ENTÃO

    INÍCIO
        MAIOR_NUM_ACIDENTES <- NUM_ACIDENTES;
        ID_CIDADE_MAIOR <- CODIGO_CIDADE;
    FIM

    TOTAL_VEICULOS <- TOTAL_VEICULOS + NUM_VEICULOS;

    SE NUM_VEICULOS < 2000 ENTÃO

    INÍCIO
        TOTAL_ACIDENTES <- TOTAL_ACIDENTES + NUM_ACIDENTES;
        CIDADES_MENOS_2000 <- CIDADES_MENOS_2000 + 1;
    FIM

FIM

MEDIA_VEICULOS <- TOTAL_VEICULOS / 5;
MEDIA_ACIDENTES <- TOTAL_ACIDENTES / CIDADES_MENOS_2000;

ESCREVA "MÉDIA DE VEÍCULOS: ", MEDIA_VEICULOS;
ESCREVA "MÉDIA DE ACIDENTES: ", MEDIA_ACIDENTES;
ESCREVA "MAIOR ÍNDICE: ", MAIOR_NUM_ACIDENTES, " CIDADE: ", ID_CIDADE_MAIOR;
ESCREVA "MENOR ÍNDICE: ", MENOR_NUM_ACIDENTES, " CIDADE: ", ID_CIDADE_MENOR;

| média aritmética | mensagem  |

| de 7 para cima   | aprovado  |

* o total de alunos aprovados;
* o total de alunos de exame;
* o total de alunos reprovados;
* a média da classe.

DECLARE NOTA, MEDIA_ALUNO, NUMERO_ALUNO, NUMERO_NOTA,
        TOTAL_APROVADOS, TOTAL_EXAME, TOTAL_REPROVADOS,

TOTAL_EXAME <- 0;

FIM

MEDIA_TURMA <- MEDIA_TURMA / 6;

ESCREVA "TOTAL DE APROVADOS: ", TOTAL_APROVADOS;
ESCREVA "TOTAL DE EXAME: ", TOTAL_EXAME;

5) Em um campeonato de futebol existem cinco times e cada um possui onze jogadores. Faça um programa que receba a idade, o peso e a altura de cada um dos jogadores, calcule e mostre:
* a quantidade de jogadores com idade inferior a 18 anos;
* a média das idades dos jogadores de cada time;
* a média das alturas de todos os jogadores do campeonato; e
* a porcentagem de jogadores com mais de 80 kg entre todos os jogadores do campeonato.

ALGORITMO

DECLARE TIME, JOGADOR, IDADE, PESO, ALTURA,
        TOTAL_MENORES_18, TOTAL_JOGADORES,
        TOTAL_ALTURA, TOTAL_PESO_MAIS_80,
        SOMA_IDADE_TIME, MEDIA_IDADE_TIME,
        MEDIA_ALTURA_CAMPEONATO, PORCENTAGEM_PESO NUMERICO;

TOTAL_MENORES_18 <- 0;
TOTAL_JOGADORES <- 0;
TOTAL_ALTURA <- 0;
TOTAL_PESO_MAIS_80 <- 0;

PARA TIME DE 1 ATÉ 5 FAÇA

INÍCIO

    SOMA_IDADE_TIME <- 0;

    PARA JOGADOR DE 1 ATÉ 11 FAÇA

    INÍCIO

        ESCREVA "INFORME A IDADE DO JOGADOR ", JOGADOR, " DO TIME ", TIME;
        LEIA IDADE;

        ESCREVA "INFORME O PESO DO JOGADOR ", JOGADOR, " DO TIME ", TIME;
        LEIA PESO;

        ESCREVA "INFORME A ALTURA DO JOGADOR ", JOGADOR, " DO TIME ", TIME;
        LEIA ALTURA;

        SOMA_IDADE_TIME <- SOMA_IDADE_TIME + IDADE;
        TOTAL_ALTURA <- TOTAL_ALTURA + ALTURA;
        TOTAL_JOGADORES <- TOTAL_JOGADORES + 1;

            TOTAL_MENORES_18 <- TOTAL_MENORES_18 + 1;

            TOTAL_PESO_MAIS_80 <- TOTAL_PESO_MAIS_80 + 1;
        FIM

    FIM

    MEDIA_IDADE_TIME <- SOMA_IDADE_TIME / 11;
    ESCREVA "MEDIA DE IDADES DO TIME ", TIME, ": ", MEDIA_IDADE_TIME;

FIM

MEDIA_ALTURA_CAMPEONATO <- TOTAL_ALTURA / TOTAL_JOGADORES;
PORCENTAGEM_PESO <- (TOTAL_PESO_MAIS_80 / TOTAL_JOGADORES) * 100;

9) Faça um programa que leia um número não determinado de pares de valores [m, n], todos inteiros e positivos, um par de cada vez, e que calcule e mostre a soma de todos os números inteiros entre m e n (inclusive). A digitação de pares terminará quando m for maior ou igual a n.

ALGORITMO

DECLARE M, N, I, SOMA NUMERICO;

REPITA

    ESCREVA "INFORME O VALOR DE M E N (COM M < N):";
    LEIA M;
    LEIA N;

    SE M < N ENTAO
        SOMA <- 0;

        PARA I DE M ATÉ N FAÇA
            SOMA <- SOMA + I;
        FIM

        ESCREVA "A SOMA DOS INTEIROS ENTRE M E N E: ", SOMA;
    FIM

ATÉ M >= N;

FIM_ALGORITMO
